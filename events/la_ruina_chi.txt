# La Ruina - China Collapse Events
# Historical basis: China was already in civil war by 1946. The difference
# in La Ruina is that American support for the KMT evaporates completely
# (US is collapsing too), and the Soviet hand in Manchuria becomes erratic
# (USSR is collapsing too). Both sides are suddenly fighting blind,
# and the warlords who had gone quiet start making noise again.

add_namespace = la_ruina_chi

# ============================================================
# CHINA EVENT 1: THE AMERICAN MISSION LEAVES
# In real history, Gen. Marshall spent 1946 trying to broker
# a KMT-CCP ceasefire. In La Ruina, he never arrives.
# The KMT realizes it is alone.
# ============================================================
country_event = {
	id = la_ruina_chi.1
	title = la_ruina_chi.1.t
	desc = la_ruina_chi.1.d
	picture = GFX_report_event_generic_sign_treaty1

	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		tag = CHI  # Nationalist China/KMT
		la_ruina_can_collapse = yes
	}

	option = {
		name = la_ruina_chi.1.a  # "The Americans have their own problems now. We are alone."

		add_stability = -0.20
		add_political_power = -150
		add_ideas = la_ruina_societal_collapse
		set_country_flag = la_ruina_chi_no_usa

		country_event = { id = la_ruina_chi.2 days = 14 random = 21 }
	}
}

# ============================================================
# CHINA EVENT 2: KMT CORRUPTION COLLAPSES THE ECONOMY
# Without American credit, KMT hyperinflation accelerates.
# Chiang's generals are looting the provinces.
# ============================================================
country_event = {
	id = la_ruina_chi.2
	title = la_ruina_chi.2.t
	desc = la_ruina_chi.2.d
	picture = GFX_report_event_gathering_protest

	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = la_ruina_chi.2.a  # "A soldier's monthly pay cannot buy a bag of rice."

		add_stability = -0.25
		add_war_support = -0.20
		set_country_flag = la_ruina_chi_hyperinflation

		# Defections to CCP
		add_popularity = { ideology = communism popularity = 15 }

		country_event = { id = la_ruina_chi.3 days = 21 random = 21 }
	}
}

# ============================================================
# CHINA EVENT 3: THE WARLORDS RETURN
# When central government fails in China, regional strongmen
# fill the void. This happened before. It is happening again.
# ============================================================
country_event = {
	id = la_ruina_chi.3
	title = la_ruina_chi.3.t
	desc = la_ruina_chi.3.d
	picture = GFX_report_event_mil_generic

	is_triggered_only = yes
	fire_only_once = yes

	# Option A: Ma Clique (Northwest Muslim warlords) declares autonomy
	option = {
		name = la_ruina_chi.3.a  # "The Ma family armies no longer answer to Nanking."
		ai_chance = { factor = 50 }

		add_stability = -0.20
		set_country_flag = la_ruina_chi_ma_clique
		la_ruina_spawn_warlord_region = yes

		country_event = { id = la_ruina_chi.4 days = 14 random = 14 }
	}

	# Option B: Yunnan governor declares independence — the Southwest goes its own way
	option = {
		name = la_ruina_chi.3.b  # "Yunnan has closed its borders to KMT troops."
		ai_chance = { factor = 50 }

		add_stability = -0.15
		set_country_flag = la_ruina_chi_yunnan_independent
		la_ruina_spawn_warlord_region = yes

		country_event = { id = la_ruina_chi.4 days = 14 random = 14 }
	}
}

# ============================================================
# CHINA EVENT 4: THE KMT BREAKS
# Chiang cannot fight the CCP, the warlords, and economic
# collapse simultaneously. Something has to give.
# ============================================================
country_event = {
	id = la_ruina_chi.4
	title = la_ruina_chi.4.t
	desc = la_ruina_chi.4.d
	picture = GFX_report_event_civil_war

	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = la_ruina_chi.4.a  # "The Nationalist government has lost the Mandate of Heaven."

		set_country_flag = la_ruina_collapsed
		remove_ideas = la_ruina_societal_collapse
		add_ideas = la_ruina_fractured_state

		# The civil war deepens — CCP gains significant territory
		start_civil_war = {
			ideology = communism
			size = 0.55   # CCP is already strong — they get a head start
		}
	}
}

# ============================================================
# CCP COUNTERPART EVENT: THE SOVIETS STOP HELPING
# Filed under CHN (Communist China) — Mao's forces also feel
# the tremors when the USSR collapses.
# ============================================================
country_event = {
	id = la_ruina_chi.10
	title = la_ruina_chi.10.t
	desc = la_ruina_chi.10.d
	picture = GFX_report_event_mil_generic

	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		tag = CHN  # Communist China
		la_ruina_can_collapse = yes
		SOV = { has_country_flag = la_ruina_sov_stalin_dead }
	}

	option = {
		name = la_ruina_chi.10.a  # "Moscow is silent. We win this ourselves, or we do not win it."

		add_stability = -0.15
		add_war_support = 0.10  # Mao's forces are ideologically motivated — this hardens them
		add_political_power = -100
		set_country_flag = la_ruina_chn_alone

		# No collapse for the CCP — they're already fighting
		# But they're weakened without Soviet supply
		add_ideas = la_ruina_societal_collapse
	}
}
